# A function with a declared return type must return a value on every path.
# These reach their end without one: the condition can be false, or the loop
# can finish without the return ever being taken.
#
# The check walks the generated quadruples rather than the source, since the
# jumps between them already describe the control flow of the body.

program partialReturns;
var r : int;

int onlyOnTrueBranch ( n : int ) [
   {  if (n > 0) {
         return n;
      };
   }
];

int missingElseBranch ( n : int ) [
   {  if (n > 0) {
         return n;
      } else {
         print("no return here");
      };
   }
];

int onlyInsideLoop ( n : int ) [
   {  do {
         if (n == 3) {
            return n;
         };
         n = n + 1;
      } while (n < 10);
   }
];

main {
    r = 0;
}
end
