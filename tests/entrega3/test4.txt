program funciones;
var n, res : int ;
    x : float;

void saluda ( veces : int )
[  var k : int;
   {  k = 0;
      do {
         print("hola desde la funcion void");
         k = k + 1;
      } while ( k < veces );
   }
];

int fact ( n : int )
[
   {  if ( n > 1 )
      {  return n * fact(n - 1); }
      else
      {  return 1; };
   }
];

float mitad ( v : int )
[
   {  return v / 2; }
];

main
{  saluda(2);
   res = fact(5);
   print("fact(5) = ", res);
   x = mitad(7);
   print("mitad(7) = ", x);
}
end
