# Caso 3: llamadas a funcion dentro de expresiones
# Verifica F_CallExpr como alternativa de Atom

program caso3;

var x, y, total : int;

int doble(n : int) [
    {
        return n * 2;
    }
];

int sumar(a : int, b : int) [
    {
        return a + b;
    }
];

main {
    x = 5;
    y = 10;

    # F_CallExpr en asignacion simple
    total = doble(x);

    # F_CallExpr combinada con aritmetica
    total = doble(x) + sumar(y, 3) * 2;

    # F_CallExpr anidada en otra llamada
    total = sumar(doble(x), doble(y));

    # F_CallExpr en condicion
    if (doble(x) > sumar(y, 1)) {
        print("doble gana");
    };

    # F_CallExpr en print
    print("resultado: ", doble(x) + sumar(y, x));
} end
