# Caso 2: funciones con tipo de retorno (int, float, void)
# Verifica ReturnType y Return_Statement

program caso2;

var a, b : int;
r : float;

int maximo(x : int, y : int) [
    {
        if (x > y) {
            return x;
        } else {
            return y;
        };
    }
];

float promedio(x : int, y : int) [
    var sum : float;
    {
        sum = x + y;
        return sum / 2.0;
    }
];

void saludar() [
    {
        print("Hola desde funcion void");
        return;
    }
];

main {
    a = 30;
    b = 45;
    print("Maximo: ", maximo(a, b));
    r = promedio(a, b);
    print("Promedio: ", r);
    saludar();
} end
