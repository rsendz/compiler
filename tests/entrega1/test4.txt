# Caso 4: anidamientos profundos
# if dentro de else, do/while con if adentro,
# funciones con variables locales y llamadas anidadas

program caso4;

var a, b, c : int;
flag : int;

int abs_val(n : int) [
    {
        if (n < 0) {
            return -n;
        } else {
            return n;
        };
    }
];

void procesar(x : int, y : int) [
    var diff : int;
    {
        diff = abs_val(x - y);

        if (diff > 10) {
            if (x > y) {
                print("x mucho mayor que y");
            } else {
                if (y > x) {
                    print("y mucho mayor que x");
                } else {
                    print("iguales pero diff > 10?");
                };
            };
        } else {
            print("diferencia pequena: ", diff);
        };
    }
];

main {
    a = 50;
    b = 3;
    c = 0;

    # do/while con if anidado
    do {
        c = c + 1;
        if (c == 3) {
            print("c llego a 3");
        };
        if (c > 4) {
            print("ya pasamos de 4");
        } else {
            print("c vale: ", c);
        };
    } while (c < 6);

    # Llamada con expresiones anidadas
    procesar(a, b);
    procesar(abs_val(-20), abs_val(b - a));

    flag = abs_val(a - b) + abs_val(b - c);
    print("flag: ", flag);
} end
