# Caso 5: expresiones con todos los operadores
# Aritmeticos: + - * /
# Relacionales: > < >= <= == !=
# Unarios: +x, -x
# Parentesis para precedencia

program caso5;

var a, b, c, d, e : int;
f : float;

main {
    a = 10;
    b = 3;

    # Aritmetica con precedencia
    c = a + b * 2;
    d = (a + b) * 2;
    e = a - b * 2 + 4 / 2;
    f = 3.14 * 2.0 + 1.5 / 0.5;

    # Signo unario
    e = -5 + +a;

    # Todos los operadores relacionales
    if (a > b) {
        print("a > b");
    };
    if (a < 100) {
        print("a < 100");
    };
    if (a >= 10) {
        print("a >= 10");
    };
    if (b <= 3) {
        print("b <= 3");
    };
    if (a == 10) {
        print("a == 10");
    };
    if (a != b) {
        print("a != b");
    };

    print("c=", c, " d=", d, " e=", e);
} end
